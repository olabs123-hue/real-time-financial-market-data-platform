cmake_minimum_required(VERSION 3.16)
project(market_data_platform LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_executable(feed_producer src/main.cpp)
target_include_directories(feed_producer PRIVATE include)
option(ENABLE_KAFKA "Enable Kafka publishing" OFF)
if(ENABLE_KAFKA)
 find_library(RDKAFKA_LIBRARY rdkafka REQUIRED)
 find_path(RDKAFKA_INCLUDE librdkafka/rdkafka.h REQUIRED)
 target_include_directories(feed_producer PRIVATE ${RDKAFKA_INCLUDE})
 target_link_libraries(feed_producer PRIVATE ${RDKAFKA_LIBRARY})
 target_compile_definitions(feed_producer PRIVATE WITH_KAFKA)
endif()
enable_testing()
add_executable(test_book tests/test_book.cpp)
target_include_directories(test_book PRIVATE include)
add_test(NAME book_tests COMMAND test_book)
